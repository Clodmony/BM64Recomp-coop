# Each machine turns its own camera with its own C-buttons, and each player's
# stick is turned by its own machine's camera on both machines (#33,
# docs/coop/design.md, *The camera*). Green Garden stage 1, room 0x29, as
# session-camera: C-left turns the camera there from yaw 90 to 45, C-right
# from 90 to 135. The client turns its camera and walks player 2; then the
# host turns its own and walks player 1; both games still run the same
# frames.
#session
#env BM64_COOP_WARP=0x29:0x13
#env BM64_COOP_LOG_PLAYERS=1
#env BM64_COOP_LOG_HASH=1
#
#check log ^coop: session lockstep from frame 0$
#check-client log ^coop: session lockstep from frame 0$
#check no-log ^coop: session (desync|lost|refused)
#check-client no-log ^coop: session (desync|lost|refused)
#check same-hashes
#check-client log ^coop: camera follows slot=4$
# the client's C-left turned its own camera, not the host's
#check camera start yaw = 90
#check-client camera start yaw = 90
#check-client camera turned yaw = 45
#check camera turned yaw = 90
# player 2 walked by the client's stick, turned by the client's camera
#check moved turned walked2 4 100
#check-client moved turned walked2 4 100
# (stick right is -z at yaw 90; at 45 it goes +x as well)
#check delta turned walked2 4 x > 100
#check-client delta turned walked2 4 x > 100
#check delta turned walked2 4 z < -100
#check-client delta turned walked2 4 z < -100
# the host's C-right turned its own camera, not the client's
#check camera hostturn yaw = 135
#check-client camera hostturn yaw = 45
# player 1 walked by the host's stick, turned by the host's camera
#check moved hostturn end 0 100
#check-client moved hostturn end 0 100
# (at yaw 135, stick right goes -x and -z)
#check delta hostturn end 0 x < -100
#check-client delta hostturn end 0 x < -100
#check delta hostturn end 0 z < -100
#check-client delta hostturn end 0 z < -100
